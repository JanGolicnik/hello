<script &>
  const bd = new Date(2004, 9, 18);
  const t = new Date();
  let age = t.getFullYear() - bd.getFullYear();
  if (
      t.getMonth() < bd.getMonth() ||
      (t.getMonth() === bd.getMonth() && t.getDate() < bd.getDate())
  )
      age--;
  return age;
</script>, iz Ljubljane, trenutno v Bežigradu.
